def git(*args):
    return subprocess.check_output(['git', '-c', 'core.hooksPath=NUL', *args])


def blobs(ref, paths):
    payload = subprocess.run(['git', '-c', 'core.hooksPath=NUL', 'cat-file', '--batch'],
                             input=''.join(f'{ref}:{p}\n' for p in paths).encode(),
                             check=True, capture_output=True).stdout
    result, cursor = {}, 0
    for path in paths:
        end = payload.index(b'\n', cursor)
        oid, kind, size = payload[cursor:end].decode().split()
        assert kind == 'blob'
        cursor = end + 1
        result[path] = payload[cursor:cursor+int(size)]
        cursor += int(size) + 1
    return result


def write_json(path, value):
    path.write_bytes((json.dumps(value, ensure_ascii=False, indent=2) + '\n').encode('utf-8'))


def archive(path, value):
    raw = (json.dumps(value, ensure_ascii=False, separators=(',', ':')) + '\n').encode('utf-8')
    path.write_bytes(gzip.compress(raw, mtime=0))
    return hashlib.sha256(raw).hexdigest()

for sn, data in discovery['snapshots'].items():
    candidates = set(data['authorization_candidates'])
    candidates.update(k for k in supplemental if k in data['nodes'])
    rows = []
    for symbol in sorted(candidates):
        n = data['nodes'][symbol]
        module = n['module']
        if module.startswith('engineering_ci.release.') or module == 'engineering_ci.cli.release':
            route = 'real-release-workflow'
        elif module == 'engineering_ci.cli.download':
            route = 'display/download conclusion only; not an authorization exit'
        elif module == 'engineering_ci.tools.performance':
            route = 'performance report vocabulary; not an authorization exit'
        elif module.startswith('engineering_ci_module.'):
            route = 'native-dagger-plan-result'
        else:
            route = module_routes.get(module.removeprefix('engineering_ci.'), 'manual review required')
        rows.append({'symbol': symbol, 'path': n['path'], 'line': n['line'],
                     'end_line': n['end_line'], 'blob': n['blob'], 'sha256': n['sha256'],
                     'discovery': 'lexical' if symbol in data['authorization_candidates'] else 'manual-entrypoint-expansion',
                     'route': route, 'resolved_runtime_callers': n['callers']})
    paths = git('ls-tree', '-r', '--name-only', data['commit']).decode().splitlines()
    search_files = [p for p in paths if (p.startswith(('src/engineering_ci/', '.dagger/src/engineering_ci_module/', 'scripts/', 'tests/', '.github/workflows/'))
                                        and p.endswith(('.py', '.yml', '.yaml'))) or p in ['pyproject.toml', 'dagger.json', '.dagger/pyproject.toml']]
    hits, source_files, dagger_entrypoints = [], [], []
    for p, raw in blobs(data['commit'], search_files).items():
        source_files.append({'path': p, 'sha256': hashlib.sha256(raw).hexdigest()})
        if p.startswith('.dagger/src/engineering_ci_module/') and p.endswith('.py'):
            for item in ast.walk(ast.parse(raw.decode('utf-8-sig'))):
                if isinstance(item, (ast.FunctionDef, ast.AsyncFunctionDef)):
                    decorators = [ast.unparse(d) for d in item.decorator_list]
                    if any(d.split('(')[0] in ['check', 'function'] for d in decorators):
                        dagger_entrypoints.append({'path':p,'line':item.lineno,'name':item.name,'decorators':decorators})
        if p.startswith(('tests/', '.github/workflows/')) or p == 'pyproject.toml':
            for i, line in enumerate(raw.decode('utf-8-sig').splitlines(), 1):
                if entry_names.search(line):
                    hits.append({'path': p, 'line': i, 'text': line.strip()})
    inventory['snapshots'][sn] = {'commit': data['commit'], 'runtime_files_parsed': data['runtime_files_parsed'],
                                'symbols_parsed': data['symbols_parsed'],
                                'lexical_candidates': len(data['authorization_candidates']),
                                'expanded_candidates': rows, 'searched_files': source_files,
                                'test_workflow_entrypoint_hits': hits,
                                'dagger_decorated_entrypoints':dagger_entrypoints}
write_json(ROOT / 'discovery-inventory.json', inventory)

graph = {'purpose': 'Issue #125 B manual data-contract map, no new authority decisions',
         'direction': 'Producer/fact -> Validator/Consumer; physical call graph is in mechanical-discovery archive',
         'edge_kinds': {'contract_input': 'API input/output compatibility to be checked, not a discovered call',
                        'static_call': 'Direct call verified in pinned code',
                        'manual_dispatch': 'Local import/closure/Workflow dispatch traced manually',
                        'unresolved': 'External source/issuer/provider/environment edge needs actual observation'},
         'nodes': {}, 'edges': []}


def node(sn, symbol):
    key = sn + ':' + symbol
    if key not in graph['nodes']:
        n = discovery['snapshots'][sn]['nodes'][symbol]
        graph['nodes'][key] = {'snapshot': sn, 'symbol': symbol,
                               **{k: n[k] for k in ['commit', 'path', 'line', 'end_line', 'blob', 'sha256']},
                               'resolved_runtime_callers': n['callers']}
    return key


def external(sn, name, observation):
    key = sn + ':external:' + name
    graph['nodes'][key] = {'snapshot': sn, 'external': name, 'resolution': 'unresolved', 'needed_observation': observation}
    return key


def edge(sn, producer, consumer, kind, fact):
    def resolve(value):
        return value if ':external:' in value else node(sn, value)
    graph['edges'].append({'id': f'E{len(graph["edges"])+1:03}', 'from': resolve(producer), 'to': resolve(consumer),
                           'kind': kind, 'fact': fact})
