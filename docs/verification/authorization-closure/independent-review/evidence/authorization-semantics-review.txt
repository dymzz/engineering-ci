RESULTS=[]
def plain(value):
    if dataclasses.is_dataclass(value):return plain(dataclasses.asdict(value))
    if isinstance(value,dict):return {str(k):plain(v) for k,v in value.items()}
    if isinstance(value,(list,tuple)):return [plain(v) for v in value]
    if isinstance(value,(set,frozenset)):return sorted(plain(v) for v in value)
    if isinstance(value,(datetime.datetime,datetime.date)):return value.isoformat()
    if isinstance(value,Path):return value.as_posix()
    if callable(value):return {'controlled_callable':getattr(value,'__module__','')+'.'+getattr(value,'__qualname__',type(value).__name__)}
    if value is None or isinstance(value,(str,int,float,bool)):return value
    return {'controlled_object_type':type(value).__module__+'.'+type(value).__name__}
def digest(value):return hashlib.sha256(json.dumps(plain(value),sort_keys=True,separators=(',',':')).encode()).hexdigest()
def status(out):return out.get('returned',{}).get('status')
def error(out,types=()):return bool(out.get('raised')) and (not types or out['raised']['type'] in types)
def has_coverage(out):return any(r.get('admitted') is True for r in out.get('returned',{}).get('accepted_coverage_by_instance',{}).values())
def run(name,family,expected,inputs,invoke,predicate,*,note=''):
    frozen=plain(inputs);before=digest(inputs)
    try:out={'returned':plain(invoke())}
    except Exception as exc:
        out={'raised':{'type':type(exc).__name__,'code':getattr(exc,'code',None),'native_code':getattr(exc,'native_code',None),'diagnostic':str(exc)}}
    try:conforms=bool(predicate(out))
    except Exception as exc:conforms=False;out['predicate_error']=type(exc).__name__+': '+str(exc)
    RESULTS.append({'case':name,'family':family,'snapshot':SNAPSHOT,'commit':manifest[SNAPSHOT]['commit'],
                    'expected':expected,'inputs':frozen,'input_sha256':before,'input_mutated':digest(inputs)!=before,
                    'actual':out,'result':'conforms' if conforms else 'counterexample','note':note,
                    'authority_scope':'Controlled normalized inputs/callback/provider fixtures only; no real issuer or production authorization acceptance.'})
def case(name,fn):
    try:fn()
    except Exception as exc:RESULTS.append({'case':name,'snapshot':SNAPSHOT,'result':'probe_error','diagnostic':type(exc).__name__+': '+str(exc)})
