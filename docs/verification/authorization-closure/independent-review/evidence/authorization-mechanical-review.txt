def owned(node):
    for child in ast.iter_child_nodes(node):
        if isinstance(child,(ast.FunctionDef,ast.AsyncFunctionDef,ast.ClassDef,ast.Lambda)):continue
        yield child
        yield from owned(child)

def absolute(module,level,name,path):
    if not level:return name or ''
    package=module if path.endswith('/__init__.py') else module.rsplit('.',1)[0]
    parts=package.split('.')
    return '.'.join(parts[:len(parts)-level+1]+([name] if name else []))

def imports(node,module,path):
    found={}
    for child in owned(node):
        if isinstance(child,ast.Import):
            for alias in child.names:found[alias.asname or alias.name.split('.')[0]]=alias.name if alias.asname else alias.name.split('.')[0]
        if isinstance(child,ast.ImportFrom):
            base=absolute(module,child.level,child.module,path)
            for alias in child.names:found[alias.asname or alias.name]=base+'.'+alias.name
    return found

output={}
for label,spec in manifest.items():
    root=Path(spec['root']);files=[];symbols={};trees={};modules={};globals_={}
    for item in spec['files']:
        path=item['path']
        if not path.endswith('.py') or not path.startswith(('src/','.dagger/src/','scripts/')):continue
        rel=(path.split('src/',1)[1] if 'src/' in path else path)[:-3].replace('/','.')
        module=rel.removesuffix('.__init__');tree=ast.parse((root/path).read_text(encoding='utf-8'),filename=path)
        trees[module]=tree;modules[module]=path;globals_[module]=imports(tree,module,path)
        files.append({'path':path,'sha256':item['sha256'],'module':module})
        symbols[module+'.__module_entry__']={'id':module+'.__module_entry__','module':module,'path':path,'line':1,'end_line':len((root/path).read_text(encoding='utf-8').splitlines()),
             'body_ast_sha256':hashlib.sha256(ast.dump(tree,include_attributes=False).encode()).hexdigest(),'node':tree,'prefix':module}
        def collect(node,prefix):
            for child in ast.iter_child_nodes(node):
                if isinstance(child,(ast.FunctionDef,ast.AsyncFunctionDef,ast.ClassDef)):
                    name=prefix+'.'+child.name
                    if not isinstance(child,ast.ClassDef):
                        symbols[name]={'id':name,'module':module,'path':path,'line':child.lineno,'end_line':child.end_lineno,
                         'body_ast_sha256':hashlib.sha256(ast.dump(child,include_attributes=False).encode()).hexdigest(),
                         'node':child,'prefix':prefix}
                    collect(child,name)
                else:collect(child,prefix)
        collect(tree,module)
    calls=[];candidates=[]
    for name,record in symbols.items():
        node=record['node'];module=record['module'];aliases={**globals_[module],**imports(node,module,record['path'])}
        params={p.arg for p in node.args.posonlyargs+node.args.args+node.args.kwonlyargs} if hasattr(node,'args') else set()
        if hasattr(node,'args') and node.args.vararg:params.add(node.args.vararg.arg)
        if hasattr(node,'args') and node.args.kwarg:params.add(node.args.kwarg.arg)
        fields=set();body=list(owned(node))
        for child in body:
            if isinstance(child,ast.Dict):fields|={k.value for k in child.keys if isinstance(k,ast.Constant) and isinstance(k.value,str)}
            if not isinstance(child,ast.Call):continue
            expression=ast.unparse(child.func);head=expression.split('.',1)[0];tail=expression[len(head):]
            target=None;kind='unresolved_receiver_or_value'
            if head in params:kind='callback_or_parameter'
            elif head in aliases:
                possible=aliases[head]+tail

            elif isinstance(child.func,ast.Name):

                    prefix=prefix.rsplit('.',1)[0]
                if target is None and expression in dir(builtins):kind='builtin'
            elif head in {'self','cls'}:
                possible=record['prefix']+tail
                if possible in symbols:target=possible;kind='resolved_same_class'
            calls.append({'caller':name,'path':record['path'],'line':child.lineno,'expression':expression,'target':target,'kind':kind,
                          'keywords':[k.arg for k in child.keywords]})
        if fields&KEYS or name in SEEDS:
            candidates.append({'id':name,'path':record['path'],'line':record['line'],'surface_keys':sorted(fields&KEYS),'seed':name in SEEDS})
    reverse={}
    for call in calls:

    summary={'files':len(files),'symbols':len(symbols),'calls':len(calls),'resolved_calls':sum(c['kind'].startswith('resolved_') for c in calls),
       'callback_calls':sum(c['kind']=='callback_or_parameter' for c in calls),'expanded_sink_candidates':len(candidates),'workflow_files':len(boundaries)}
