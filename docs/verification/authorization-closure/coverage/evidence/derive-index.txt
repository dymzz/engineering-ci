def git(*args):

def write(path, data):
    path.write_bytes((json.dumps(data,ensure_ascii=False,indent=2)+'\n').encode('utf-8'))


def qualified(name):
    return name if name.startswith('engineering_ci') else 'engineering_ci.'+name


def numbers(text):
    result=[]
    for part in text.split():
        ends=[int(x) for x in part.split('-')]
        result.extend(f'E{i:03}' for i in range(ends[0],ends[-1]+1))
    return result

def function_ref(sn,symbol):
    symbol=qualified(symbol)

        n=mechanical['snapshots'][sn]['nodes'][symbol]

blob_cache={}
def blob(ref,path):
    key=(ref,path)
    if key not in blob_cache: blob_cache[key]=git('show',ref+':'+path)
    return blob_cache[key]

def file_ref(ref,path,kind):
    key=ref+':'+path
    if key not in catalog:
        raw=blob(ref,path)
        catalog[key]={'kind':kind,'commit':ref,'path':path,'sha256':hashlib.sha256(raw).hexdigest(),
                      'blob':hashlib.sha1(b'blob '+str(len(raw)).encode()+b'\0'+raw).hexdigest(),
                      'url':f'https://github.com/dymzz/engineering-ci/blob/{ref}/{path}'}
    return key

def excerpt(ref,start,end):
    source=catalog[ref]
    lines=blob(source['commit'],source['path']).decode('utf-8-sig').splitlines()
    assert 1<=start<=end<=len(lines)
    text='\n'.join(lines[start-1:end])+'\n'
    return {'source_ref':ref,'start':start,'end':end,'text':text,
            'text_sha256':hashlib.sha256(text.encode('utf-8')).hexdigest(),
            'normalization':'UTF-8/LF excerpt; file SHA remains the exact Git blob content'}


def expand_proof(sn,spec,edge):
    assert spec['edge']==edge['id'] and spec['proof_object'] and spec['scope_limit']
    assert set(spec['layers'])<={'component_contract','handoff_contract'}
    source_steps=[]
    for step in spec['source_steps']:
        ref=function_ref(sn,step['symbol']);source=catalog[ref]
        assert source['line']<=step['start']<=step['end']<=source['end_line'],(spec['id'],step)
        source_steps.append({**excerpt(ref,step['start'],step['end']),
                             'roles':step['roles'],'proves':step['proves']})
    test=spec['test'];file=Path(test['file']).stem
    ref=test_ref(sn,file,test['name']);source=catalog[ref]
    tree=ast.parse(blob(source['commit'],source['path']).decode('utf-8-sig'))
    case=next(n for n in ast.walk(tree) if isinstance(n,(ast.FunctionDef,ast.AsyncFunctionDef))
              and n.name==test['name'] and n.lineno==source['line'])
    assertions=[]
    for binding in test['assertions']:
        statement=next(n for n in ast.walk(case) if isinstance(n,ast.Assert) and n.lineno==binding['line'])
        assertions.append({**excerpt(ref,statement.lineno,statement.end_lineno),
                           'assertion_expression':ast.unparse(statement.test),
                           'roles':binding['roles'],'proves':binding['proves']})
    invocations=[]
    for line in test['act_lines']:
        calls=[n for n in ast.walk(case) if isinstance(n,ast.Call) and n.lineno==line]
        assert calls,(spec['id'],line)
        call=max(calls,key=lambda n:(n.end_lineno-n.lineno,n.end_col_offset-n.col_offset))
        invocations.append({**excerpt(ref,call.lineno,call.end_lineno),'expression':ast.unparse(call.func)})
    contexts=[{**excerpt(ref,c['start'],c['end']),'proves':c['proves']} for c in test['fixture_context']]
    return {'id':spec['id'],'layers':spec['layers'],'proof_object':spec['proof_object'],
            'edge_binding':{'edge':edge['id'],'snapshot':sn,'from_node':edge['from'],'to_node':edge['to']},
            'method':spec['method'],'source_steps':source_steps,'test_ref':ref,
            'invocations':invocations,'assertion_bindings':assertions,'fixture_context':contexts,
            'controlled_actor':spec['controlled_actor'],'scope_limit':spec['scope_limit'],
            'recorded_suite_ref':ci_by_snapshot[sn],
            'execution_scope':'Existing fixed test source and suite receipt; no new per-case execution, origin authentication or D replay.'}
